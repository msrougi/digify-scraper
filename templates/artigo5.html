Image Scraping com Python: Guia Prático com Exemplos

📅 Janeiro 2025 · por Digify Scrapper · ⏱ 10 min

Python é a linguagem mais popular para web scraping. Com as bibliotecas certas, você cria um image scraper poderoso em poucas linhas. Neste guia, vamos do básico ao avançado.

Instalação das dependências

pip install requests beautifulsoup4 pillow

Scraper básico: imagens de uma página

import requests
from bs4 import BeautifulSoup
from urllib.parse import urljoin
from pathlib import Path

def baixar_imagens(url, pasta="imagens"):
    Path(pasta).mkdir(exist_ok=True)
    r = requests.get(url, headers={"User-Agent": "Mozilla/5.0"})
    soup = BeautifulSoup(r.text, "html.parser")
    
    for i, img in enumerate(soup.find_all("img")):
        src = img.get("src") or img.get("data-src")
        if not src:
            continue
        img_url = urljoin(url, src)
        try:
            img_r = requests.get(img_url, timeout=10)
            ext = img_url.split(".")[-1].split("?")[0][:4]
            with open(f"{pasta}/imagem_{i}.{ext}", "wb") as f:
                f.write(img_r.content)
            print(f"Baixada: {img_url}")
        except Exception as e:
            print(f"Erro: {e}")

baixar_imagens("https://exemplo.com")

Scraper avançado: site inteiro

from collections import deque

def scraper_completo(url_inicial, profundidade=2):
    visitadas = set()
    fila = deque([(url_inicial, 0)])
    imagens = set()
    dominio = url_inicial.split("/")[2]
    
    while fila:
        url, nivel = fila.popleft()
        if url in visitadas or nivel > profundidade:
            continue
        visitadas.add(url)
        
        try:
            r = requests.get(url, timeout=10,
                headers={"User-Agent": "Mozilla/5.0"})
            soup = BeautifulSoup(r.text, "html.parser")
            
            # Coleta imagens
            for img in soup.find_all("img"):
                for attr in ["src", "srcset", "data-src"]:
                    src = img.get(attr, "")
                    if src:
                        imagens.add(urljoin(url, src.split(",")[0].strip()))
            
            # Segue links internos
            for a in soup.find_all("a", href=True):
                link = urljoin(url, a["href"])
                if dominio in link and link not in visitadas:
                    fila.append((link, nivel + 1))
                    
        except Exception as e:
            print(f"Erro em {url}: {e}")
    
    return list(imagens)

urls = scraper_completo("https://exemplo.com", profundidade=2)
print(f"{len(urls)} imagens encontradas")

Boas práticas de scraping

  • Sempre adicione um User-Agent nos headers
  • Respeite o robots.txt do site
  • Adicione um delay entre requisições (time.sleep(0.5))
  • Não sobrecarregue servidores de terceiros
  • Verifique os Termos de Serviço do site
